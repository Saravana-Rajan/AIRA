## Global Constraints

- **Name:** the product name in user-facing copy is **AIRA**. The code prefix is `aira`. The demo shop id is `murugan_dairy`.
- **Spatial coordinates:**
  - Points are `[y, x]`.
  - Boxes are `[ymin, xmin, ymax, xmax]`.
  - Both are integers normalized to 0–1000.
- **Audio:**
  - Client → server: 16 kHz mono PCM16, base64.
  - Server → client: 24 kHz mono PCM16, base64.
- **Language and verbosity:** languages are `"ta" | "hi" | "en"`. Verbosity is `"short" | "detail"`.
- **Model IDs appear ONLY in `services/live/aira_live/models.py`:**

  | Role | Model |
  |---|---|
  | `live` | `gemini-3.8-live` |
  | `point` | `gemini-robotics-er-2-preview` |
  | `count` | `gemini-robotics-er-2-preview` |
  | `read` | `gemini-3.8-flash` |
  | `verify` | `gemini-3.8-flash` |

  `fallback_for("point"|"count")` returns `gemini-3.8-flash`.
- **GCP:**
  - Region `asia-south1`, which covers Cloud Run and Firestore and gives low latency for Indian users.
  - Service `aira-live` with `--min-instances 1 --timeout 3600 --session-affinity --cpu-boost`.
- **Secrets:**
  - `GEMINI_API_KEY`, `WHATSAPP_TOKEN`, `WHATSAPP_PHONE_ID`, `WHATSAPP_VERIFY_TOKEN`, `TELEGRAM_TOKEN`.
  - They live only in Secret Manager or a local `.env`, and are never committed.
- **The model proposes, the backend commits:** every consequential tool returns `confirm.propose(...)` first. Commits happen only through `confirm.resolve(...)`.
- **Contract changes:** one PR that updates the TS types, the pydantic mirror and the fixtures together, reviewed by Saravana.
- **Demo video:** at most 4:00. The rules text says "3 to 4 minutes", but the submission form's field label says "up to 3 minutes". The team chose 4 minutes; confirm this with the organiser.

## Review Focus

1. **Unknown or future message type:**
   - The server answers `{"t":"error","code":"unsupported"}` and keeps the socket open (test in Task 4).
   - The app ignores unknown server messages (test in Task 5).
2. **Oversized frame:** a base64 JPEG over 1.5 MB is rejected with `unsupported` and never stored (test in Task 4).
3. **Fixture drift:** every fixture validates in BOTH the TS guard and the pydantic adapter, and the fixtures cover every message type (tests in Tasks 1 and 2).
4. **Double confirmation:** resolving the same `confirm_id` twice commits exactly once. The second call returns the stored result (test in Task 3).
5. **Missing secrets:** without `GEMINI_API_KEY`, `aira-live` starts in mock mode and `/healthz` reports `"mock": true` (test in Task 4).

```ts
import {
  CLIENT_EVENT_NAMES, CONFIRM_KINDS, CUE_NAMES, ENTITIES, ERROR_CODES, FRAME_PURPOSES, LANGS, VERBOSITIES,
  type ConfirmKind, type CueName, type Entity, type ErrorCode, type FramePurpose, type Lang, type Verbosity,
} from "./enums";

export type ClientMsg =
  | { t: "hello"; uid: string; idToken?: string; shopId: string; lang: Lang; verbosity: Verbosity; appVersion: string; resume?: string }
  | { t: "audio"; pcm16: string }
  | { t: "frame"; id: string; jpeg: string; w: number; h: number; ts: number; purpose: FramePurpose }
  | { t: "text"; text: string }
  | { t: "event"; name: (typeof CLIENT_EVENT_NAMES)[number]; data: Record<string, unknown> }
  | { t: "confirm"; id: string; answer: "yes" | "no"; value?: string }
  | { t: "stop" };

export type ServerMsg =
  | { t: "ready"; sessionId: string; resume: string; mock: boolean }
  | { t: "audio"; pcm24: string }
  | { t: "transcript"; role: "user" | "aira"; text: string; final: boolean }
  | { t: "request_frame"; purpose: FramePurpose; burst?: { count: number; intervalMs: number }; continuous?: boolean; stop?: boolean }
  | { t: "target"; frameId: string; label: string; point: [number, number]; box?: [number, number, number, number]; confidence: number }
  | { t: "count"; sku: string; counted: number; expected: number | null; final: boolean; confidence: number }
  | { t: "state"; entity: Entity; id: string; state: string; summary: string }
  | { t: "confirm"; id: string; kind: ConfirmKind; prompt: string }
  | { t: "cue"; cue: CueName }
  | { t: "error"; code: ErrorCode; say: string };

export const CLIENT_MSG_TYPES = ["hello", "audio", "frame", "text", "event", "confirm", "stop"] as const;
export const SERVER_MSG_TYPES = ["ready", "audio", "transcript", "request_frame", "target", "count", "state", "confirm", "cue", "error"] as const;

const isObj = (x: unknown): x is Record<string, any> => typeof x === "object" && x !== null && !Array.isArray(x);
const isStr = (x: unknown): x is string => typeof x === "string";
const isNum = (x: unknown): x is number => typeof x === "number" && Number.isFinite(x);
const inList = (list: readonly string[], x: unknown) => isStr(x) && list.includes(x);
const isNorm = (n: unknown) => isNum(n) && n >= 0 && n <= 1000;
const isPoint = (p: unknown) => Array.isArray(p) && p.length === 2 && p.every(isNorm);
const isBox = (b: unknown) => Array.isArray(b) && b.length === 4 && b.every(isNorm);

export function isClientMsg(x: unknown): x is ClientMsg {
  if (!isObj(x) || !inList(CLIENT_MSG_TYPES, x.t)) return false;
  switch (x.t) {
    case "hello": return isStr(x.uid) && isStr(x.shopId) && inList(LANGS, x.lang) && inList(VERBOSITIES, x.verbosity) && isStr(x.appVersion);
    case "audio": return isStr(x.pcm16);
    case "frame": return isStr(x.id) && isStr(x.jpeg) && isNum(x.w) && isNum(x.h) && isNum(x.ts) && inList(FRAME_PURPOSES, x.purpose);
    case "text": return isStr(x.text);
    case "event": return inList(CLIENT_EVENT_NAMES, x.name) && isObj(x.data);
    case "confirm": return isStr(x.id) && (x.answer === "yes" || x.answer === "no") && (x.value === undefined || isStr(x.value));
    case "stop": return true;
  }
  return false;
}

export function isServerMsg(x: unknown): x is ServerMsg {
  if (!isObj(x) || !inList(SERVER_MSG_TYPES, x.t)) return false;
  switch (x.t) {
    case "ready": return isStr(x.sessionId) && isStr(x.resume) && typeof x.mock === "boolean";
    case "audio": return isStr(x.pcm24);
    case "transcript": return (x.role === "user" || x.role === "aira") && isStr(x.text) && typeof x.final === "boolean";
    case "request_frame": return inList(FRAME_PURPOSES, x.purpose) &&
      (x.burst === undefined || (isObj(x.burst) && isNum(x.burst.count) && isNum(x.burst.intervalMs)));
    case "target": return isStr(x.frameId) && isStr(x.label) && isPoint(x.point) && (x.box === undefined || isBox(x.box)) && isNum(x.confidence);
    case "count": return isStr(x.sku) && isNum(x.counted) && (x.expected === null || isNum(x.expected)) && typeof x.final === "boolean" && isNum(x.confidence);
    case "state": return inList(ENTITIES, x.entity) && isStr(x.id) && isStr(x.state) && isStr(x.summary);
    case "confirm": return isStr(x.id) && inList(CONFIRM_KINDS, x.kind) && isStr(x.prompt);
    case "cue": return inList(CUE_NAMES, x.cue);
    case "error": return inList(ERROR_CODES, x.code) && isStr(x.say);
  }
  return false;
}
```

```python
# aira_live/frames.py
import asyncio
from collections import deque
from dataclasses import dataclass

@dataclass(frozen=True)
class Frame:
    id: str
    jpeg_b64: str
    w: int
    h: int
    ts: float
    purpose: str

class FrameStore:
    def __init__(self, maxlen: int = 30):
        self._buf: deque[Frame] = deque(maxlen=maxlen)
        self._cond = asyncio.Condition()

    def add(self, frame: Frame) -> None:
        self._buf.append(frame)
        try:
            asyncio.get_running_loop().create_task(self._notify())
        except RuntimeError:
            pass

    async def _notify(self) -> None:
        async with self._cond:
            self._cond.notify_all()

    def latest(self, n: int = 1, purpose: str | None = None) -> list[Frame]:
        items = [f for f in self._buf if purpose is None or f.purpose == purpose]
        return items[-n:]

    async def wait_for(self, count: int, purpose: str, after_ts: float, timeout_s: float) -> list[Frame]:
        def ready() -> list[Frame]:
            return [f for f in self._buf if f.purpose == purpose and f.ts > after_ts]
        try:
            async with asyncio.timeout(timeout_s):
                async with self._cond:
                    await self._cond.wait_for(lambda: len(ready()) >= count)
        except TimeoutError:
            return []
        return ready()[:count]
```

```python
        burst = Burst(count=count, interval_ms=interval_ms) if count > 1 else None

        return await self.frames.wait_for(count=count, purpose=purpose, after_ts=since, timeout_s=timeout_s)
```

```python
# aira_live/confirm.py
"""Model proposes, backend commits. Every consequential tool returns propose(...); commits only via resolve(...)."""
import uuid
from typing import Awaitable, Callable, Literal, Protocol
from aira_live import config
from aira_live.contracts import ConfirmMsg
from aira_live.i18n import t
from aira_live.session import get_session

Committer = Callable[[dict, str | None], Awaitable[dict]]
_COMMITTERS: dict[str, Committer] = {}
_KINDS = {"order", "payment", "sale", "discrepancy", "generic"}


class ProposalStore(Protocol):
    async def put(self, shop_id: str, confirm_id: str, doc: dict) -> None: ...
    async def get(self, shop_id: str, confirm_id: str) -> dict | None: ...
    async def update(self, shop_id: str, confirm_id: str, fields: dict) -> None: ...


class MemoryProposalStore:
    def __init__(self):
        self._docs: dict[tuple[str, str], dict] = {}

    async def put(self, shop_id, confirm_id, doc):
        self._docs[(shop_id, confirm_id)] = dict(doc)

    async def get(self, shop_id, confirm_id):
        d = self._docs.get((shop_id, confirm_id))
        return dict(d) if d else None

    async def update(self, shop_id, confirm_id, fields):
        self._docs[(shop_id, confirm_id)].update(fields)


class FirestoreProposalStore:
    def _ref(self, shop_id, confirm_id):
        from aira_live.db import db
        return db().collection("shops").document(shop_id).collection("proposals").document(confirm_id)

    async def put(self, shop_id, confirm_id, doc):
        await self._ref(shop_id, confirm_id).set(doc)

    async def get(self, shop_id, confirm_id):
        snap = await self._ref(shop_id, confirm_id).get()
        return snap.to_dict() if snap.exists else None

    async def update(self, shop_id, confirm_id, fields):
        await self._ref(shop_id, confirm_id).update(fields)


_store: ProposalStore | None = None

def set_store(store: ProposalStore) -> None:
    global _store
    _store = store

def store() -> ProposalStore:
    global _store
    if _store is None:
        _store = MemoryProposalStore() if config.load().mock else FirestoreProposalStore()
    return _store

def register_committer(name: str, fn: Committer) -> None:
    _COMMITTERS[name] = fn

async def propose(kind: str, prompt: str, payload: dict, committer: str) -> dict:
    s = get_session()
    cid = uuid.uuid4().hex[:10]
    await store().put(s.shop_id, cid, {"kind": kind, "prompt": prompt, "payload": payload,
                                       "committer": committer, "status": "pending", "sessionId": s.session_id})
    await s.send(ConfirmMsg(id=cid, kind=kind if kind in _KINDS else "generic", prompt=prompt))
    return {"needs_confirm": True, "confirm_id": cid, "say": prompt}

async def resolve(confirm_id: str, answer: Literal["yes", "no"], value: str | None = None) -> dict:
    s = get_session()
    doc = await store().get(s.shop_id, confirm_id)
    if doc is None:
        return {"say": t("confirm.unknown", s.lang), "error": "unknown_confirm"}
    if doc["status"] != "pending":
        return {"say": t("confirm.already", s.lang), "status": doc["status"], "result": doc.get("result")}
    if answer == "no":
        await store().update(s.shop_id, confirm_id, {"status": "rejected"})
        return {"say": t("confirm.cancelled", s.lang), "status": "rejected"}
    result = await _COMMITTERS[doc["committer"]](doc["payload"], value)
    await store().update(s.shop_id, confirm_id, {"status": "committed", "result": result})
    return {"status": "committed", **result}
```

```python
# aira_live/main.py
import json
from fastapi import FastAPI, WebSocket, WebSocketDisconnect
from pydantic import ValidationError
from aira_live import config, confirm, sessions
from aira_live.contracts import (ConfirmReplyMsg, CueMsg, ErrorMsg, FrameMsg, HelloMsg, ReadyMsg, StopMsg,
                                 TextMsg, TranscriptMsg, client_adapter, server_adapter)
from aira_live.frames import Frame
from aira_live.i18n import t
from aira_live.session import Session, current_session

app = FastAPI(title="aira-live")

@app.get("/healthz")
def healthz():
    return {"ok": True, "mock": config.load().mock}

@app.websocket("/ws")
async def ws_endpoint(ws: WebSocket):
    await ws.accept()
    settings = config.load()

    async def send(msg):
        exclude_none = getattr(msg, "t", None) != "count"
        await ws.send_text(server_adapter.dump_json(msg, by_alias=True, exclude_none=exclude_none).decode())

    session: Session | None = None
    lang = "en"
    try:
        while True:
            raw = await ws.receive_text()
            try:
                data = json.loads(raw)
                if data.get("t") == "frame" and len(data.get("jpeg", "")) > settings.max_frame_b64:
                    await send(ErrorMsg(code="unsupported", say=t("error.too_large", lang)))
                    continue
                msg = client_adapter.validate_python(data)
            except (ValidationError, ValueError):
                await send(ErrorMsg(code="unsupported", say=t("error.unsupported", lang)))
                continue

            if isinstance(msg, HelloMsg):
                session = Session(msg.uid, msg.shop_id, msg.lang, msg.verbosity, send)
                lang = msg.lang
                current_session.set(session)
                sessions.register(session)
                await send(ReadyMsg(session_id=session.session_id, resume=session.session_id, mock=settings.mock))
            elif isinstance(msg, FrameMsg) and session:
                session.frames.add(Frame(id=msg.id, jpeg_b64=msg.jpeg, w=msg.w, h=msg.h, ts=msg.ts, purpose=msg.purpose))
            elif isinstance(msg, TextMsg):
                await send(TranscriptMsg(role="aira", text=t("mock.echo", lang, text=msg.text), final=True))
            elif isinstance(msg, ConfirmReplyMsg) and session:
                result = await confirm.resolve(msg.id, msg.answer, msg.value)
                await send(TranscriptMsg(role="aira", text=result.get("say", ""), final=True))
            elif isinstance(msg, StopMsg):
                await send(CueMsg(cue="done"))
    except WebSocketDisconnect:
        pass
    finally:
        if session:
            sessions.unregister(session)
```

```python
"""Run one Live tool without the phone.
Usage: uv run python -m aira_live.devtools.call_tool <tool> --args '<json>' [--image a.jpg ...] [--purpose count] [--shop murugan_dairy] [--lang en]
Prints every ServerMsg the tool sends (one JSON per line) and then the tool's return dict."""
import argparse
import asyncio
import base64
import json
import sys
import time
from pathlib import Path
from aira_live.contracts import server_adapter
from aira_live.frames import Frame
from aira_live.session import Session, current_session
from aira_live.tools import TOOLS_BY_NAME

async def _run(name: str, args: dict, images: list[str], purpose: str, shop: str, lang: str) -> dict:
    async def send(msg):
        print(server_adapter.dump_json(msg, by_alias=True).decode())
    s = Session(uid="devtools", shop_id=shop, lang=lang, verbosity="short", send=send)
    current_session.set(s)
    now = time.time() * 1000
    for i, p in enumerate(images):
        b64 = base64.b64encode(Path(p).read_bytes()).decode()
        s.frames.add(Frame(id=f"img{i}", jpeg_b64=b64, w=0, h=0, ts=now + i + 1, purpose=purpose))
    return await TOOLS_BY_NAME[name](**args)

def main(argv: list[str] | None = None) -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("tool", choices=sorted(TOOLS_BY_NAME))
    ap.add_argument("--args", default="{}")
    ap.add_argument("--image", action="append", default=[])
    ap.add_argument("--purpose", default="ask")
    ap.add_argument("--shop", default="murugan_dairy")
    ap.add_argument("--lang", default="en")
    ns = ap.parse_args(argv)
    result = asyncio.run(_run(ns.tool, json.loads(ns.args), ns.image, ns.purpose, ns.shop, ns.lang))
    print(json.dumps(result, ensure_ascii=False))

if __name__ == "__main__":
    main(sys.argv[1:])
```

```ts
import { isServerMsg, type ClientMsg, type ServerMsg } from "@aira/contracts";

export const backoffMs = (attempt: number) => Math.min(500 * 2 ** attempt, 15000);
type Hello = Extract<ClientMsg, { t: "hello" }>;
type Handler = (m: ServerMsg) => void;

export class AiraSocket {
  private ws: WebSocket | null = null;
  private handlers = new Set<Handler>();
  private attempt = 0;
  private hello: Hello | null = null;
  private closedByUser = false;
  private WS: typeof WebSocket;

  constructor(private url: string, opts: { WebSocketImpl?: typeof WebSocket } = {}) {
    this.WS = opts.WebSocketImpl ?? WebSocket;
  }

  connect(hello: Hello) {
    this.hello = hello;
    this.closedByUser = false;
    const ws = new this.WS(this.url);
    this.ws = ws;
    ws.onopen = () => { this.attempt = 0; ws.send(JSON.stringify(this.hello)); };
    ws.onmessage = (e: MessageEvent | { data: string }) => {
      let m: unknown;
      try { m = JSON.parse(String(e.data)); } catch { return; }
      if (isServerMsg(m)) this.handlers.forEach((h) => h(m as ServerMsg));
    };
    ws.onclose = () => {
      if (this.closedByUser || !this.hello) return;
      setTimeout(() => this.connect(this.hello!), backoffMs(this.attempt++));
    };
  }

  send(msg: ClientMsg): boolean {
    if (!this.ws || this.ws.readyState !== 1) return false;
    this.ws.send(JSON.stringify(msg));
    return true;
  }

  on(h: Handler) { this.handlers.add(h); return () => this.handlers.delete(h); }
  close() { this.closedByUser = true; this.ws?.close(); }
}
```

```ts
// src/stock.ts
import type { InventoryDoc, Product } from "@aira/contracts";
export type StockRow = { sku: string; name: string; onHand: number; unit: string; low: boolean };

export function toRows(products: Product[], inventory: InventoryDoc[]): StockRow[] {
  const bySku = new Map(inventory.map((i) => [i.sku, i.onHand]));
  return products
    .map((p) => {
      const onHand = bySku.get(p.sku) ?? 0;
      return { sku: p.sku, name: p.name, onHand, unit: p.unit, low: onHand <= p.reorderThreshold };
    })
    .sort((a, b) => a.name.localeCompare(b.name));
}
```

```tsx
import { useEffect, useState } from "preact/hooks";
import { collection, onSnapshot } from "firebase/firestore";
import type { InventoryDoc, Product } from "@aira/contracts";
import { db } from "./firebase";
import { toRows, type StockRow } from "./stock";

export function StockTable({ shopId }: { shopId: string }) {
  const [products, setProducts] = useState<Product[]>([]);
  const [inventory, setInventory] = useState<InventoryDoc[]>([]);
  useEffect(() => {
    const u1 = onSnapshot(collection(db, `shops/${shopId}/products`), (s) => setProducts(s.docs.map((d) => d.data() as Product)));
    const u2 = onSnapshot(collection(db, `shops/${shopId}/inventory`), (s) => setInventory(s.docs.map((d) => d.data() as InventoryDoc)));
    return () => { u1(); u2(); };
  }, [shopId]);
  const rows: StockRow[] = toRows(products, inventory);
  return (
    <table aria-label="Live stock">
      <thead><tr><th>Product</th><th>On hand</th><th>Status</th></tr></thead>
      <tbody>
        {rows.map((r) => (
          <tr key={r.sku}><td>{r.name}</td><td>{r.onHand} {r.unit}</td><td>{r.low ? "Reorder" : "OK"}</td></tr>
        ))}
      </tbody>
    </table>
  );
}
```

```python
        docs.append((f"{base}/products/{p['sku']}", dict(p)))
        docs.append((f"{base}/inventory/{p['sku']}", {"sku": p["sku"], "onHand": 0, "batches": [], "updatedAt": now}))
    for s in catalog["suppliers"]:
        doc = {k: v for k, v in s.items() if k != "phoneEnv"}
        doc["phone"] = env.get(s["phoneEnv"], "")
        docs.append((f"{base}/suppliers/{s['id']}", doc))
    return docs

def seed(catalog_path: Path, project: str) -> int:
    from google.cloud import firestore
    client = firestore.Client(project=project)
    docs = build_docs(json.loads(catalog_path.read_text(encoding="utf-8")), dict(os.environ))
    for path, data in docs:
        client.document(path).set(data)
    return len(docs)

if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--project", required=True)
    ap.add_argument("--catalog", default=str(ROOT / "content" / "catalog" / "murugan_dairy.json"))
    ns = ap.parse_args()
    print(f"seeded {seed(Path(ns.catalog), ns.project)} docs")
```

```bash
#!/usr/bin/env bash
# Usage: PROJECT=<gcp-project-id> ./infra/bootstrap.sh     (needs: gcloud auth login, billing enabled, firebase login)
set -euo pipefail
: "${PROJECT:?set PROJECT}"
REGION=asia-south1
gcloud config set project "$PROJECT"

gcloud services enable run.googleapis.com artifactregistry.googleapis.com cloudbuild.googleapis.com \
  secretmanager.googleapis.com firestore.googleapis.com pubsub.googleapis.com bigquery.googleapis.com \
  cloudscheduler.googleapis.com storage.googleapis.com logging.googleapis.com monitoring.googleapis.com \
  generativelanguage.googleapis.com firebase.googleapis.com firebasehosting.googleapis.com \
  firebaseremoteconfig.googleapis.com fcm.googleapis.com

gcloud firestore databases create --location="$REGION" --type=firestore-native || true

PN=$(gcloud projects describe "$PROJECT" --format='value(projectNumber)')
SA="${PN}-compute@developer.gserviceaccount.com"
for s in GEMINI_API_KEY WHATSAPP_TOKEN WHATSAPP_PHONE_ID WHATSAPP_VERIFY_TOKEN TELEGRAM_TOKEN; do
  gcloud secrets create "$s" --replication-policy=automatic 2>/dev/null || true
  if [ -n "${!s:-}" ]; then printf '%s' "${!s}" | gcloud secrets versions add "$s" --data-file=- >/dev/null; fi
  gcloud secrets add-iam-policy-binding "$s" --member="serviceAccount:${SA}" --role=roles/secretmanager.secretAccessor >/dev/null
done
gcloud projects add-iam-policy-binding "$PROJECT" --member="serviceAccount:${SA}" --role=roles/datastore.user >/dev/null

gcloud run deploy aira-live --source services/live --region "$REGION" --allow-unauthenticated \
  --min-instances 1 --timeout 3600 --session-affinity --cpu-boost --memory 1Gi \
  --set-secrets GEMINI_API_KEY=GEMINI_API_KEY:latest,WHATSAPP_TOKEN=WHATSAPP_TOKEN:latest,WHATSAPP_PHONE_ID=WHATSAPP_PHONE_ID:latest,WHATSAPP_VERIFY_TOKEN=WHATSAPP_VERIFY_TOKEN:latest,TELEGRAM_TOKEN=TELEGRAM_TOKEN:latest \
  --set-env-vars GCP_PROJECT="$PROJECT"
URL=$(gcloud run services describe aira-live --region "$REGION" --format='value(status.url)')
echo "aira-live: $URL"
curl -fsS "$URL/healthz" && echo

npx firebase-tools hosting:sites:create "${PROJECT}-dashboard" --project "$PROJECT" || true
npx firebase-tools deploy --only firestore:rules,firestore:indexes --project "$PROJECT"

npm run build -w apps/dashboard
npx firebase-tools deploy --only hosting --project "$PROJECT"
```
